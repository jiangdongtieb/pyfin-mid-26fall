def fetch(secid, fqt, name, tries=4):
    params = {"secid": secid, "fields1": FIELDS1, "fields2": FIELDS2,
              "klt": "101", "fqt": fqt, "beg": BEG, "end": END}
    for i in range(tries):
        try:
            r = requests.get(URL, params=params, headers=HEAD, timeout=30)
            r.raise_for_status()
            data = r.json().get("data")
            if not data or not data.get("klines"):
                log("  %s 空（第%d次）" % (name, i + 1))
                time.sleep(6)
                continue
            rows = [k.split(",") for k in data["klines"]]
            df = pd.DataFrame(rows, columns=COLS)
            df["名称"] = data.get("name")
            df["代码"] = data.get("code")
            df["复权方式"] = {"0": "不复权", "1": "前复权", "2": "后复权"}.get(str(fqt), str(fqt))
            for c in COLS[1:10]:
                df[c] = pd.to_numeric(df[c], errors="coerce")
            df.to_csv(RAW + "\\" + name + ".csv", index=False, encoding="utf-8-sig")
            log("  保存 %s.csv  %d 行  %s ~ %s" % (name, len(df), df["日期"].iloc[0], df["日期"].iloc[-1]))
            return df
        except Exception as e:
            log("  %s 异常 %s（第%d次）" % (name, str(e)[:100], i + 1))
            time.sleep(8)
    log("  !! %s 失败" % name)
    return None

2.3 为什么我们自己重建复权价（本章最重要的一节）
Task1 原文：Adjust prices for corporate actions (后复权) at least whenever you compute returns — with more than ten years of data, unadjusted prices are unusable (comparing 前复权 and 后复权 is a bonus).
牧原上市 12 年做过 5 次转增股本：2015 年 10 转 10、2016 年 10 转 10、2018 年 10 转 8、2020 年 10 转 7、2021 年 10 转 4，外加 14 次现金分红。不复权价格在每个除权日都会「假摔」：10 转 10 那天股价直接腰斩，图上看起来像崩盘，其实是白送了一倍股数。我们用不复权价算出来的收益率，在这 14 天里最大的假跌幅是 -50.91%。
复权的原理其实就一句话：把「分红 + 送股」当成投资者的真实所得，倒推出一个连续的价格序列。 交易所每天公布的「除权除息参考价」就是现成的桥梁：
复权公式
除权除息参考价  P_ref(t) = ( P(t-1) - 每股现金红利 ) / ( 1 + 每股转增比例 )
真实日收益率(含红利再投资)  r(t) = P(t) / P_ref(t) - 1
后复权因子  F(t) = F(t-1) * P(t-1) / P_ref(t)      # 非除权日 P_ref = P(t-1)，因子不变
后复权价  = P(t) * F(t)      # 以上市首日为基准 1，因子只增不减
前复权价  = P(t) * F(t) / F(最后一天)   # 以今天为基准，方便和当前股价对照
PanelPrep.build_adjust_factor()：真实代码
    # ---------- 2.2 自建复权因子 ----------
    def build_adjust_factor(self, px):
        dv = self.raw["fhps_002714"].copy()
        dv["ex_dt"] = dv["EX_DIVIDEND_DATE"].dt.normalize()
        dv["dps"] = dv["PRETAX_BONUS_RMB"] / 10.0                       # 每股现金红利
        dv["bonus_ratio"] = dv["BONUS_IT_RATIO"].fillna(0) / 10.0       # 每股送转股
        dv = dv.sort_values("ex_dt")

        ref_price = px["close"].shift(1)      # 除权除息参考价，默认 = 昨收盘
        detail = []
        for _, row in dv.iterrows():
            d = row["ex_dt"]
            if d < px.index.min() or d > px.index.max():
                continue
            if d not in px.index:             # 除权日撞上非交易日就顺延
                loc = px.index.searchsorted(d)
                if loc >= len(px.index):
                    continue
                d = px.index[loc]
            pos = px.index.get_loc(d)
            if pos == 0:
                continue
            p_prev = px["close"].iloc[pos - 1]
            p_ref = (p_prev - row["dps"]) / (1.0 + row["bonus_ratio"])
            ref_price.loc[d] = p_ref
            detail.append({"除权除息日": d, "昨收盘": p_prev, "每股派息": row["dps"],
                           "每股转增": row["bonus_ratio"], "除权除息参考价": p_ref,
                           "方案": row["IMPL_PLAN_PROFILE"]})
        self.note("处理除权除息事件 %d 个（含 5 次转增股本）" % len(detail))
        factor = (px["close"].shift(1) / ref_price).fillna(1.0)
        factor.iloc[0] = 1.0
        factor = factor.cumprod()
        self.dividend_detail = pd.DataFrame(detail)
        return factor, ref_price
逐行看懂这段：
dv['dps'] = dv['PRETAX_BONUS_RMB'] / 10.0：东财给的分红是「每 10 股派 X 元」，要除以 10 变成每股。BONUS_IT_RATIO 是「每 10 股转增 X 股」，同样除以 10，并用 fillna(0) 把没有转增的分红年份补成 0（否则整列变 NaN）。
ref_price = px['close'].shift(1)：先把除权除息参考价默认设成昨收盘（shift(1) 往前挪一格，Class02 讲的基本操作），然后只对除权日那几天用 ref_price.loc[d] = p_ref 覆盖。先给全体一个默认值、再改少数特例，比逐日判断快得多。
if d not in px.index：除权日可能撞上周末或节假日（不是交易日），这时用 searchsorted(d) 找到它在交易日序列里的插入位置，顺延到下一个交易日处理。牧原 14 次除权里有几次是这种情况。
ratio = (px['close'].shift(1) / ref_price).fillna(1.0)：非除权日分子分母相等，比值就是 1；除权日比值 > 1。factor = ratio.cumprod() 用累乘得到复权因子——cumprod 是「累计乘积」，正好把每次除权的调整幅度连乘起来。
最后 adj_close = close_raw * factor：向量化的一行乘法（Class02 讲的 broadcasting 思想），不用写循环。
