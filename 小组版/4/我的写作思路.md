build_sample()
    def build_sample(self):
        px = self.px
        # 最慢的指标是 MA250 / 250日动量 -> 预热期 250 个交易日
        warm = 250
        px = px.iloc[warm:].copy()
        # 未来收益
        for h in self.horizons:
            px["fwd_ret_%d" % h] = px["close_adj"].shift(-h) / px["close_adj"] - 1.0
        px["fwd_ret_21"] = px["close_adj"].shift(-21) / px["close_adj"] - 1.0     # 约 1 个月
        px["fwd_ret_63"] = px["close_adj"].shift(-63) / px["close_adj"] - 1.0     # 约 3 个月
        # 剔除一字涨跌停（当天无法成交）
        px["tradable"] = (px["is_one_word_board"] == 0).astype(int)
        self.sample = px
        self.note("样本区间 %s ~ %s，共 %d 个交易日（预热期剔除 %d 天，一字板 %d 天标记为不可成交）"
                  % (px.index.min().date(), px.index.max().date(), len(px), warm,
                     int((px["is_one_word_board"] == 1).sum())))
        # 年交易日数按样本自己校准
        yrs = (px.index[-1] - px.index[0]).days / 365.25
        self.trading_days = int(round(len(px) / yrs))
        self.note("样本实测年交易日数 %.0f（用于年化）" % self.trading_days)
        return px

6.3 择时回测：T+1、一字板顺延、交易成本（最容易被追问的细节）
backtest()
    def backtest(self, signal, name, allow_defer=True):
        px = self.sample
        sig = pd.Series(signal, index=px.index).fillna(0).clip(0, 1)
        if allow_defer:
            # 一字板当天不能成交：持仓状态维持前一天的值
            pos = sig.copy()
            locked = px["is_one_word_board"] == 1
            for i in range(1, len(pos)):
                if locked.iloc[i]:
                    pos.iloc[i] = pos.iloc[i - 1]
            sig_exec = pos
        else:
            sig_exec = sig
        position = sig_exec.shift(1).fillna(0)      # t-1 收盘决策，t 日持有
        strat_ret = position * px["ret"]
        turnover = position.diff().abs().fillna(position.abs())
        cost = np.where(position.diff() > 0, self.BUY_COST,
                        np.where(position.diff() < 0, self.SELL_COST, 0.0))
        cost = pd.Series(cost, index=px.index).fillna(0) * turnover
        strat_net = strat_ret - cost
        nav = (1 + strat_net).cumprod()
        bh = (1 + px["ret"]).cumprod()
        m = self.metrics(strat_net, nav, name)
        m["换手次数"] = int((turnover > 0).sum())
        m["持仓天数占比%"] = round(float(position.mean() * 100), 2)
        m["交易成本累计%"] = round(float(cost.sum() * 100), 3)
        m["买入持有年化%"] = round(float((bh.iloc[-1]) ** (self.trading_days / len(px)) - 1) * 100, 2)
        m["买入持有最大回撤%"] = round(float(self.max_drawdown(bh) * 100), 2)
        m["买入持有年化波动%"] = round(float(px["ret"].std() * np.sqrt(self.trading_days) * 100), 2)
        m["买入持有净值"] = round(float(bh.iloc[-1]), 3)
这一段是回测的工程核心，逐行拆：
position = sig_exec.shift(1).fillna(0)：T+1。t-1 日收盘算出信号，t 日才持有——shift(1) 让信号晚一天生效。不用 shift 的话，你等于「用今天的收盘信号在今天收盘买入」，这在现实中做不到（收盘价是收盘那一刻才知道的）。
if locked.iloc[i]: pos.iloc[i] = pos.iloc[i-1]：一字板当天持仓状态沿用前一天。一字板全天只有一个价格，挂单根本成交不了，所以信号在当天不可执行，顺延到下一个可交易日。这个循环虽然慢（2807 次），但语义精确。
cost = np.where(position.diff() > 0, BUY, np.where(position.diff() < 0, SELL, 0))：diff() 计算持仓变化，>0 是买入（付佣金万 2.5）、<0 是卖出（佣金 + 印花税千 5）、=0 是持有不动（无成本）。np.where 嵌套是向量化三选一。
strat_net = strat_ret - cost，nav = (1 + strat_net).cumprod()：策略每日净收益累乘成净值曲线。
绩效指标：年化 = nav.iloc[-1]  (241/样本天数) - 1（241 是样本实测年交易日数）；波动 = ret.std() * sqrt(241)；最大回撤 = min((nav - nav.cummax())/nav.cummax())——cummax 是滚动最大值（Class02）。
【坑】年化交易日数不要写死 252。A 股实际年交易日约 241（节假日多），我们直接用样本「天数 ÷ 年数」自校准。写死 252 会让年化收益虚高约 4%。这个细节很多同学会翻车。

newey_west()：用 statsmodels 做 HAC
    def newey_west(self, both):
        rows = []
        try:
            import statsmodels.api as sm
            for h in self.horizons + [21, 63]:
                d = both[["composite", "fwd_ret_%d" % h]].dropna()
                if len(d) < 200:
                    continue
                y = d["fwd_ret_%d" % h].values
                x = sm.add_constant((d["composite"].values - 50) / 50.0)
                ols = sm.OLS(y, x).fit(cov_type="HAC", cov_kwds={"maxlags": h - 1})
                ols_iid = sm.OLS(y, x).fit()
                rows.append({"持有期": "%d日" % h, "斜率(每+50分)": round(float(ols.params[1]) * 100, 3),
                             "NeweyWest_t": round(float(ols.tvalues[1]), 3),
                             "普通OLS_t": round(float(ols_iid.tvalues[1]), 3),
                             "样本数": len(d)})
        except Exception as e:
            print("  Newey-West 计算失败：%s" % str(e)[:120])
        return pd.DataFrame(rows)
为什么要 Newey-West（加分项里的重点）：未来 20 日收益在相邻两天高度重叠——1 月 1 日买的 20 日收益和 1 月 2 日买的 20 日收益，有 19 天是同一个行情，不是独立观测。普通 OLS 的 t 值会把这种「假独立」当成有效信息，t 值虚高。sm.OLS(y, x).fit(cov_type='HAC', cov_kwds={'maxlags': h - 1}) 就是 Newey-West 异方差自相关稳健标准误，滞后阶取 h-1（20 日收益重叠 19 天）。我们实测：20 日斜率的 t 从普通 OLS 的 1.566 掉到 HAC 的 0.480——「看起来有效」的信号其实不显著。这句话是本项目统计严谨性的最高点，答辩必问，务必背熟。
【坑】statsmodels 在环境里装好了，但 cov_kwds={'maxlags': h-1} 里的 maxlags 必须是持有期减 1：60 日收益的滞后要 59，写 60 是错的。另外 HAC 在小样本下会偏保守，我们顺带报告了普通 OLS 的 t 做对照，让老师看到两个口径的差距，而不是只给一个数。

# 第 7 章　三个交付文件是怎么从一份代码生成的
这一章讲工程实现（占分 15% 的 Reproducibility, code quality 部分），回答三个问题：为什么只写一份主代码？为什么 notebook 是生成的不是手写的？报告和 PPT 的数字从哪里来？
7.1 主程序 muyuan_core.py：一份代码，两种跑法
muyuan_core.py 是全项目的唯一代码源，约 1100 行，分 6 个 SECTION：
文件结构
# SECTION 1: DataLoader          读原始 CSV
# SECTION 2: PanelPrep            清洗 + 复权 + 市值 + 涨跌停
# SECTION 3: TechnicalFramework   Task1 技术指标与打分
# SECTION 4: FundamentalFramework Task2 基本面 + PIT + 六维度
# SECTION 5: TimingBacktester     Task3 检验 + 回测 + 评级预测
# SECTION 6: stage_data/stage_task1/2/3  主流程（4 个 stage）
两种跑法：
当脚本跑：python code/muyuan_core.py，if __name__ == '__main__': run_all() 依次执行 4 个 stage，打印全部日志并把 27 个结果文件写进 output/。
当 notebook 跑：build_notebook.py 用正则把每个 SECTION 切成一个代码单元，插上讲解用的 markdown，再调用 stage_data() 等 4 个函数分步执行——不是复制粘贴代码，是切同一份文件，所以两边的代码永远一致。
【提示】_find_base() 函数处理了路径问题：从 code/ 目录跑时用 __file__ 定位项目根；在 Jupyter 里 __file__ 不存在（这是 Python 与 Jupyter 的一个经典差异），就退回 os.getcwd()。不写这个，notebook 里一跑就 ModuleNotFoundError 或 FileNotFoundError。
7.2 build_notebook.py：怎么把 .py 变成 .ipynb
原理：.ipynb 就是个 JSON 文件，cells 列表里每个单元要么是 markdown（source 是文本），要么是 code（source 是代码行、outputs 是执行结果）。所以：
切块：用正则把 # SECTION n: 标记切成段落，parts[1]~parts[6] 就是 6 个代码单元。
排版：26 个单元 = 15 个 markdown（封面、Task0 调研、方法说明、结果解读、加分项自查）+ 11 个 code（1 个环境配置 + 5 个类定义 + 1 个 stage 函数 + 4 个执行单元）。
执行：最后用 jupyter nbconvert --to notebook --execute --inplace 把全部代码跑一遍，输出写死进文件。交上去的 notebook 打开就能看到图和表，不需要重跑。
【坑】在 Jupyter 里跑我们的代码，第一行 sys.stdout = io.TextIOWrapper(sys.stdout.buffer, ...) 会直接报 AttributeError: 'OutStream' object has no attribute 'buffer'——因为 Jupyter 的 stdout 是 ipykernel 的 OutStream，没有 .buffer。修复就是加 if hasattr(sys.stdout, 'buffer'): 判断（这个坑我们真踩了，第一次 nbconvert 就挂在第一行）。
7.3 报告与 PPT 的数字为什么不会抄错
报告（build_report.py）的每个数字都从 output/*.csv 或 run_summary.json 用 pd.read_csv(...) 读出来再填进 Word 表格，正文里的关键结论也是格式化字符串拼接（例如用 %.2f%% 这种格式符把回测年化收益填进句子）。没有一处是手敲的数字——手抄 27 个文件的上百个数字，必抄错几个，而老师核对数据时错一个都会显得不严谨。run_summary.json 把最关键的数字汇总成一个文件，报告和 PPT 都从它读。
PPT（build_ppt.py）同理：标题、要点、图表、建议卡全部从 json/csv 读。改一个回测参数 → 重跑 muyuan_core.py → 重跑三个生成器，全部交付文件自动更新。这就是「可复现性」在工程上的落地。
